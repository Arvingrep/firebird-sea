DROP TABLE IF EXISTS `#@__transactionlogs`;
CREATE TABLE `#@__transactionlogs` (`id` BIGINT(10) NOT NULL AUTO_INCREMENT, `userid` INT(8) NOT NULL DEFAULT '0' COMMENT '会员ID', `amout` DECIMAL(8,2) NOT NULL DEFAULT '0.00' COMMENT '金额', `type` INT(1) NOT NULL DEFAULT '0' COMMENT '操作类型', `note` VARCHAR(255) NOT NULL DEFAULT '' COMMENT '备注', `pubdate` INT(10) NOT NULL DEFAULT '0' COMMENT '操作时间',PRIMARY KEY (`id`), KEY `userid` (`userid`)) ENGINE = MyISAM DEFAULT CHARSET = utf8 COMMENT = '消费记录';
DROP TABLE IF EXISTS `#@__updatemessage`;
CREATE TABLE `#@__updatemessage` (`id` BIGINT(10) NOT NULL AUTO_INCREMENT, `module` VARCHAR(50) NOT NULL DEFAULT '' COMMENT '模块', `part` VARCHAR(50) NOT NULL DEFAULT '' COMMENT '二级说明', `param` TEXT NULL COMMENT '参数', `uid` INT(10) NOT NULL DEFAULT '0' COMMENT '用户id', `notify` VARCHAR(255) NOT NULL DEFAULT '' COMMENT '标题', `config` TEXT NULL COMMENT '参数', `is_update` INT(1) NOT NULL DEFAULT '0' COMMENT '0-未通知,1-已通知', `type` INT(1) NOT NULL DEFAULT '0' COMMENT '0-后台通知,1-前台通知', `online` INT(1) NOT NULL DEFAULT '0' COMMENT '1-用户直接通知', `time` INT(10) NOT NULL DEFAULT '0' COMMENT '时间',PRIMARY KEY (`id`), KEY `type` (`type`), KEY `module` (`module`), KEY `part` (`part`), KEY `uid` (`uid`), KEY `is_update` (`is_update`), KEY `online` (`online`), KEY `time` (`time`)) ENGINE = MyISAM DEFAULT CHARSET = utf8mb4 COMMENT = '消息通知队列';
